create table profiles ( id uuid references auth.users primary key, first_name text, last_name text, avatar_url text, created_at timestamptz default now() ); create table groups ( id uuid primary key default gen_random_uuid(), name text not null, days int[] not null, time text, created_at timestamptz default now() ); create table students ( id uuid primary key default gen_random_uuid(), group_id uuid references groups(id) on delete cascade, first_name text, last_name text, middle_name text, iin text, phone text, parent_phone text, created_at timestamptz default now() ); create table attendance ( id uuid primary key default gen_random_uuid(), student_id uuid references students(id) on delete cascade, date date not null, status text default 'none', unique(student_id, date) ); create table payments ( id uuid primary key default gen_random_uuid(), student_id uuid references students(id) on delete cascade, year int, month int, paid boolean default false, unique(student_id, year, month) ); alter table profiles enable row level security; alter table groups enable row level security; alter table students enable row level security; alter table attendance enable row level security; alter table payments enable row level security; create policy "auth" on profiles for all using (auth.role()='authenticated'); create policy "auth" on groups for all using (auth.role()='authenticated'); create policy "auth" on students for all using (auth.role()='authenticated'); create policy "auth" on attendance for all using (auth.role()='authenticated'); create policy "auth" on payments for all using (auth.role()='authenticated');
